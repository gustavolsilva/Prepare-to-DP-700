T-SQL vs. PySpark vs. KQL: Sintaxe Comparada para Engenharia de Dados (Domínio 2)

Sintaxe Comparada dos Padrões Mais Exigidos no DP-700

Operação T-SQL (Data Warehouse) PySpark (Lakehouse) KQL (Eventhouse)
Filtro e Seleção 
SELECT id, total
FROM Sales
WHERE status = 'Active';
 
df.select("id", "total") \
  .filter(col("status") == "Active")
 
Sales
| where status == "Active"
| project id, total
Agregação por Grupo 
SELECT region, SUM(total)
FROM Sales
GROUP BY region;
 
df.groupBy("region") \
  .agg(sum("total"))
 
Sales
| summarize TotalVendas = sum(total) 
  by region
Último Registro / Linha Mais Recente 
SELECT * FROM (
  SELECT *, ROW_NUMBER() OVER(
    PARTITION BY meter_id 
    ORDER BY ts DESC) as rn
  FROM Logs
) t WHERE rn = 1;
 
w = Window.partitionBy("meter_id") \
          .orderBy(col("ts").desc())
df.withColumn("rn", row_number().over(w)) \
  .filter(col("rn") == 1)
 
Logs
| summarize arg_max(ts, *) by meter_id
Extração de JSON e Expansão 
SELECT 
  JSON_VALUE(raw_txt, '$.sensor.temp')
FROM Telemetry;
 
df.withColumn("parsed", 
    from_json(col("raw_txt"), schema)) \
  .select("parsed.sensor.temp")
 
Telemetry
| extend obj = parse_json(raw_txt)
| project temp = obj.sensor.temp
Upsert / Carga Incremental 
MERGE INTO target t
USING source s ON t.id = s.id
WHEN MATCHED THEN UPDATE ...
WHEN NOT MATCHED THEN INSERT ...;
 
targetTable.alias("t").merge(
  sourceDF.alias("s"), "t.id = s.id"
).whenMatchedUpdate(...)
 .whenNotMatchedInsert(...)
 .execute()
